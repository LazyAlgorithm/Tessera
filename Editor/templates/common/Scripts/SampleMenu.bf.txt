using System;
using Tessera;

namespace {NAMESPACE};

/// Code-behind for UI/sample_menu.tui. Put this and a UI Document on the same object.
/// Methods named in onclick="..." / onload="..." are called by the document; fields named in
/// bind="..." are shown (and, for inputs, edited) by it.
[Component]
class SampleMenu : Component
{
	public String playerName = new .("Player") ~ delete _;
	[Range(0, 1)]
	public float volume = 0.6f;

	UIDocument Document => GetComponent<UIDocument>();

	void SetStatus(StringView text)
	{
		let status = Document.Find("status");
		if (status != null)
			status.SetText(text);
	}

	public void OnOpen()
	{
		SetStatus("Menu loaded.");
	}

	public void OnPlay()
	{
		SetStatus(scope $"Have fun, {playerName}!");
		Log.Info(scope $"SampleMenu: play pressed by {playerName}");
	}

	public void OnLouder() => volume = Mathf.Clamp01(volume + 0.1f);
	public void OnQuieter() => volume = Mathf.Clamp01(volume - 0.1f);

	/// A handler may take the element that was pressed.
	public void OnClose(UIElement button)
	{
		let screen = Document.Find("screen");
		if (screen != null)
			screen.visible = false;
	}
}
