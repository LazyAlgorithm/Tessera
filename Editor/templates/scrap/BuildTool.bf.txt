using System;
using System.Collections;
using Tessera;

namespace {NAMESPACE};

/// Building and driving controls, attached to the camera.
///   1-8 pick a part (five block materials, wheel, thruster, seat), left click places it on the face you aim at (or starts a new creation
///   on the ground), X removes a part, E sits in or leaves a seat. Driving: W/S, A/D, Space for thrusters.
/// Every action is a request through GameNet, so it works offline and in multiplayer alike.
[Component]
class BuildTool : Component
{
	public float reach = 6.0f;
	/// For automated checks: sit in the nearest vehicle and drive forward.
	public bool autoDrive = false;

	PartType mSelected = .Plastic;
	float mCameraYaw;
	float mRetry;


	public override void Start()
	{
		GameNet.AutoDrive = autoDrive;
	}

	Creation FindSeatNear(Vector3 position)
	{
		Creation best = null;
		float bestDistance = 6.5f;
		for (let component in Scene.Components)
		{
			if (let creation = component as Creation)
			{
				let distance = (creation.Transform.Position - position).Length;
				if (creation.HasSeat && !creation.HasDriver && distance < bestDistance)
				{
					best = creation;
					bestDistance = distance;
				}
			}
		}
		return best;
	}

	public override void Update()
	{
		if (MainMenu.Active)
			return;
		for (int32 i < PartInfo.Count)
		{
			if (Input.GetKeyDown((KeyCode)((int32)KeyCode.Alpha1 + i)))
				mSelected = (PartType)i;
		}
		// The mouse wheel also cycles through the hotbar.
		if (Input.MouseWheel != 0)
			mSelected = (PartType)(((int32)mSelected + (Input.MouseWheel < 0 ? 1 : PartInfo.Count - 1)) % PartInfo.Count);

		let localId = GameNet.LocalId;
		let driving = GameNet.FindDriven(localId);
		if (driving != null)
		{
			if (Input.GetKeyDown(.E))
				GameNet.Request(.Leave, driving.NetId, .(0, 0, 0), .Plastic, .Zero, 0);
			return;
		}
		let player = GameNet.FindPlayer(localId);
		if (player == null || !player.ActiveSelf)
			return;

		mRetry -= Time.DeltaTime;
		if (Input.GetKeyDown(.E) || (autoDrive && mRetry <= 0))
		{
			mRetry = 1.0f;
			let seat = FindSeatNear(player.Transform.Position);
			if (seat != null)
			{
				mCameraYaw = seat.Transform.EulerAngles.y;
				GameNet.Request(.Enter, seat.NetId, .(0, 0, 0), .Plastic, .Zero, 0);
			}
			return;
		}

		let place = Input.GetMouseButtonDown(0) && !GUI.MouseOverControl;
		let remove = Input.GetKeyDown(.X);
		if (!place && !remove)
			return;
		let direction = Transform.Forward;
		let origin = player.Transform.Position + Vector3(0, 1.0f, 0) + direction * 1.0f;
		RaycastHit hit;
		if (!Scene.Physics.Raycast(origin, direction, reach, out hit))
			return;
		let creation = hit.gameObject != null ? hit.gameObject.GetComponent<Creation>() : null;
		if (creation != null)
		{
			// Step a little off the surface: outwards finds the empty cell to fill, inwards the part to remove.
			let probe = hit.point + direction * (remove ? 0.03f : -0.03f);
			let local = creation.Transform.InverseTransformPoint(probe) / PartInfo.Grid;
			let cell = Int3((int32)Math.Round(local.x), (int32)Math.Round(local.y), (int32)Math.Round(local.z));
			GameNet.Request(remove ? .Remove : .Place, creation.NetId, cell, mSelected, .Zero, 0);
		}
		else if (place)
		{
			let motor = player.GetComponent<CharacterMotor>();
			GameNet.Request(.New, 0, .(0, 0, 0), mSelected, hit.point + Vector3(0, PartInfo.Grid * 0.5f + 0.05f, 0), motor != null ? motor.Yaw : 0);
		}
	}

	public override void LateUpdate()
	{
		// While driving, the camera trails the vehicle; on foot FollowCamera on the player handles it.
		let driving = MainMenu.Active ? null : GameNet.FindDriven(GameNet.LocalId);
		if (driving == null)
			return;
		let target = driving.Transform.Position + Vector3(0, 1.0f, 0);
		mCameraYaw += Mathf.WrapAngle(driving.Transform.EulerAngles.y - mCameraYaw) * Math.Min(1, Time.DeltaTime * 3);
		let rotation = Quaternion.Euler(18, mCameraYaw, 0);
		Transform.Position = target - rotation * Vector3.Forward * 7.0f;
		Transform.Rotation = rotation;
	}

	public override void OnGUI()
	{
		if (MainMenu.Active)
			return;
		let driving = GameNet.FindDriven(GameNet.LocalId);
		let width = 88.0f;
		let startX = GUI.Width * 0.5f - width * PartInfo.Count * 0.5f;
		let y = GUI.Height - 58;
		for (int32 i < PartInfo.Count)
		{
			let selected = (int32)mSelected == i;
			GUI.Box(startX + i * width + 3, y, width - 6, 44, selected ? .(0.20f, 0.46f, 0.72f, 0.9f) : .(0, 0, 0, 0.5f));
			GUI.Label(startX + i * width + 12, y + 6, scope $"{i + 1}");
			GUI.Label(startX + i * width + 12, y + 22, PartInfo.Name((PartType)i));
		}
		GUI.Box(12, 12, 330, driving != null ? 62 : 80, .(0, 0, 0, 0.45f));
		if (driving != null)
		{
			GUI.Label(22, 20, scope $"Driving  {(int)(driving.Speed * 3.6f)} km/h");
			GUI.Label(22, 38, "W/S drive, A/D steer, Space thrusters, E get out");
		}
		else
		{
			GUI.Label(22, 20, "Left click: place part   X: remove part");
			GUI.Label(22, 38, "1-8 or mouse wheel: choose part   E: sit in a seat");
			GUI.Label(22, 56, "WASD move, hold right mouse to look, Space jump");
		}
		let network = Scene.Network;
		if (network != null && network.Mode != .Offline)
			GUI.Label(GUI.Width - 230, 20, network.IsServer ? scope $"Hosting: {network.ClientCount} player(s) joined" : scope $"Connected as player {network.LocalClientId}");
	}
}
